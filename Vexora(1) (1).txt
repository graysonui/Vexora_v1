--// STATE + CLEANUP MANAGER
local Vexora = {
    Connections = {},
    CleanupTasks = {},
    Tabs = {},
    Widgets = {},
    ThemeName = CONFIG.DefaultTheme,
    Theme = Themes[CONFIG.DefaultTheme],
    ActiveTab = nil,
    Destroyed = false,
}

function Vexora:Connect(signal, callback, scope)
    if self.Destroyed then return nil end
    local connection = signal:Connect(callback)
    local bucket = scope or self.Connections
    table.insert(bucket, connection)
    return connection
end

function Vexora:AddCleanup(callback)
    if typeof(callback) == "function" then
        table.insert(self.CleanupTasks, callback)
    end
end

function Vexora:DisconnectBucket(bucket)
    for i = #bucket, 1, -1 do
        local connection = bucket[i]
        if typeof(connection) == "RBXScriptConnection" then
            pcall(function() connection:Disconnect() end)
        end
        bucket[i] = nil
    end
end

function Vexora:DisconnectAll()
    self:DisconnectBucket(self.Connections)
    for _, widget in ipairs(self.Widgets) do
        if widget and widget.Destroy then
            pcall(function() widget:Destroy() end)
        end
    end
    table.clear(self.Widgets)
    for i = #self.CleanupTasks, 1, -1 do
        pcall(self.CleanupTasks[i])
        self.CleanupTasks[i] = nil
    end
end

--// UI HELPERS
local function create(className, props, parent)
    local object = Instance.new(className)
    for key, value in pairs(props or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function corner(parent, radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius or 10) }, parent)
end

local function stroke(parent, color, transparency)
    return create("UIStroke", {
        Color = color or Vexora.Theme.Border,
        Transparency = transparency or 0,
        Thickness = 1,
    }, parent)
end

local function tween(object, duration, properties)
    local info = TweenInfo.new(duration or 0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
    local animation = TweenService:Create(object, info, properties)
    animation:Play()
    return animation
end

        BackgroundTransparency = 1,

        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
        Size = UDim2.new(1, 0, 0, 22),
    }, parent)
end

    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    local widthScale = viewport.X / 720
    local heightScale = viewport.Y / 500
    MainScale.Scale = math.clamp(math.min(widthScale, heightScale), 0.62, 1)

--// DRAGGING (mouse + touch)
do
    local dragging = false
    local dragInput, dragStart, startPosition
    Vexora:Connect(Topbar.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = Main.Position
            dragInput = input
        end
    end)
    Vexora:Connect(Topbar.InputChanged, function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    Vexora:Connect(UserInputService.InputChanged, function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPosition.X.Scale, startPosition.X.Offset + delta.X,
                startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
            )
        end
    end)
    Vexora:Connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

    local label = makeLabel(toast, tostring(message), 12, self.Theme.Text)
    label.Position = UDim2.fromOffset(14, 5)
    label.Size = UDim2.new(1, -22, 1, -10)
    label.TextWrapped = true
    toast.BackgroundTransparency = 1
    tween(toast, 0.18, { BackgroundTransparency = 0.04 })
    task.delay(3.2, function()
        if toast.Parent then
            tween(toast, 0.18, { BackgroundTransparency = 1 })
            task.wait(0.2)
            if toast.Parent then toast:Destroy() end
        end
    end)
end

--// THEME REGISTRY
local ThemeObjects = {}
function Vexora:RegisterThemeObject(object, property, key)
    table.insert(ThemeObjects, { Object = object, Property = property, Key = key })
end

function Vexora:ApplyTheme(themeName)
    local selected = Themes[themeName]
    if not selected then
        self:Notify("Unknown theme: " .. tostring(themeName), "error")
        return
    end
    self.ThemeName = themeName
    self.Theme = selected
    for _, item in ipairs(ThemeObjects) do
        if item.Object and item.Object.Parent and selected[item.Key] then
            pcall(function()
                item.Object[item.Property] = selected[item.Key]
            end)
        end
    end
    self:Notify("Theme changed to " .. themeName, "success")
end

function Vexora:CreateTab(name, iconText)
    local tabName = tostring(name)
    if self.Tabs[tabName] then return self.Tabs[tabName] end

    local button = create("TextButton", {
        Name = tabName .. "Tab",
        Text = (iconText and (iconText .. "  ") or "") .. tabName,
        Font = Enum.Font.GothamMedium,
        TextSize = 12,
        TextColor3 = self.Theme.Muted,
        TextXAlignment = Enum.TextXAlignment.Left,
        BackgroundColor3 = self.Theme.Surface,
        Size = UDim2.new(1, 0, 0, 38),
        AutoButtonColor = false,
    }, Sidebar)
    corner(button, 8)

    local page = create("ScrollingFrame", {
        Name = tabName .. "Page",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ScrollBarThickness = 3,
        ScrollBarImageColor3 = self.Theme.Accent,
        CanvasSize = UDim2.new(),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        Visible = false,
    }, PageHost)
    create("UIListLayout", {
        Padding = UDim.new(0, 10),
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, page)
    create("UIPadding", {
        PaddingTop = UDim.new(0, 4),
        PaddingLeft = UDim.new(0, 3),
        PaddingRight = UDim.new(0, 8),
        PaddingBottom = UDim.new(0, 10),
    }, page)

    local tab = {
        Name = tabName,
        Button = button,
        Page = page,
        Connections = {},
        Widgets = {},
    }

    function tab:Destroy()
        Vexora:DisconnectBucket(self.Connections)
        for _, widget in ipairs(self.Widgets) do
            if widget.Destroy then pcall(function() widget:Destroy() end) end
        end
        if self.Button then self.Button:Destroy() end
        if self.Page then self.Page:Destroy() end
    end

    self.Tabs[tabName] = tab
    table.insert(self.Widgets, tab)

    self:Connect(button.MouseButton1Click, function()
        self:SetActiveTab(tabName)
    end, tab.Connections)

    if not self.ActiveTab then
        self:SetActiveTab(tabName)
    end
    return tab
end

function Vexora:SetActiveTab(name)
    local tab = self.Tabs[name]
    if not tab then return end
    self.ActiveTab = name
    for tabName, item in pairs(self.Tabs) do
        local active = tabName == name
        item.Page.Visible = active
        tween(item.Button, 0.15, {
            BackgroundColor3 = active and self.Theme.Accent or self.Theme.Surface,
            TextColor3 = active and Color3.new(1, 1, 1) or self.Theme.Muted,
        })
    end
end

function Vexora:GetCurrentPage()
    local tab = self.ActiveTab and self.Tabs[self.ActiveTab]
    return tab and tab.Page
end

local function addWidgetHandle(parent, instance, connections)
    local handle = { Instance = instance, Connections = connections or {} }
    function handle:Destroy()
        Vexora:DisconnectBucket(self.Connections)
        if self.Instance then self.Instance:Destroy() end
    end
    table.insert(Vexora.Widgets, handle)
    return handle
end

function Vexora:CreateToggle(section, labelText, defaultValue, callback)
    local value = defaultValue == true
    local row = create("TextButton", {
        Text = "",
        BackgroundColor3 = self.Theme.Surface2,
        Size = UDim2.new(1, 0, 0, 38),
        AutoButtonColor = false,
    }, section.Body)
    corner(row, 8)
    local label = makeLabel(row, tostring(labelText), 12, self.Theme.Text)
    label.Position = UDim2.fromOffset(10, 0)
    label.Size = UDim2.new(1, -58, 1, 0)
    local indicator = create("Frame", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -10, 0.5, 0),
        Size = UDim2.fromOffset(34, 18),
        BackgroundColor3 = value and self.Theme.Accent or self.Theme.Border,
    }, row)
    corner(indicator, 9)
    local dot = create("Frame", {
        Position = value and UDim2.new(1, -16, 0, 2) or UDim2.fromOffset(2, 2),
        Size = UDim2.fromOffset(14, 14),
        BackgroundColor3 = Color3.new(1, 1, 1),
    }, indicator)
    corner(dot, 7)
    local function set(nextValue, silent)
        value = nextValue == true
        tween(indicator, 0.12, { BackgroundColor3 = value and self.Theme.Accent or self.Theme.Border })
        tween(dot, 0.12, { Position = value and UDim2.new(1, -16, 0, 2) or UDim2.fromOffset(2, 2) })
        if not silent and callback then callback(value) end
    end
    local connections = {}
    self:Connect(row.MouseButton1Click, function() set(not value) end, connections)
    local handle = addWidgetHandle(row, row, connections)
    handle.Set = set
    handle.Get = function() return value end
    set(value, true)
    return handle
end

function Vexora:CreateSlider(section, labelText, minimum, maximum, defaultValue, callback)
    minimum, maximum = tonumber(minimum) or 0, tonumber(maximum) or 100
    assert(maximum > minimum, "Slider maximum must be greater than minimum")
    local value = math.clamp(tonumber(defaultValue) or minimum, minimum, maximum)
    local frame = create("Frame", {
        BackgroundColor3 = self.Theme.Surface2,
        Size = UDim2.new(1, 0, 0, 58),
    }, section.Body)
    corner(frame, 8)
    local label = makeLabel(frame, tostring(labelText), 12, self.Theme.Text)
    label.Position = UDim2.fromOffset(10, 4)
    label.Size = UDim2.new(1, -70, 0, 20)
    local valueLabel = makeLabel(frame, tostring(value), 11, self.Theme.Accent)
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    valueLabel.Position = UDim2.new(1, -55, 0, 4)
    valueLabel.Size = UDim2.fromOffset(45, 20)
    local track = create("Frame", {
        Position = UDim2.new(0, 10, 0, 34),
        Size = UDim2.new(1, -20, 0, 8),
        BackgroundColor3 = self.Theme.Border,
    }, frame)
    corner(track, 5)
    local fill = create("Frame", {
        Size = UDim2.new((value - minimum) / (maximum - minimum), 0, 1, 0),
        BackgroundColor3 = self.Theme.Accent,
    }, track)
    corner(fill, 5)
    local dragging = false
    local function setFromX(x, silent)
        local alpha = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        value = math.floor((minimum + (maximum - minimum) * alpha) * 100 + 0.5) / 100
        fill.Size = UDim2.new(alpha, 0, 1, 0)
        valueLabel.Text = tostring(value)
        if not silent and callback then callback(value) end
    end
    local connections = {}
    self:Connect(track.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end, connections)
    self:Connect(UserInputService.InputChanged, function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end, connections)
    self:Connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end, connections)
    local handle = addWidgetHandle(frame, frame, connections)
    handle.Set = function(nextValue, silent)
        value = math.clamp(tonumber(nextValue) or minimum, minimum, maximum)
        local alpha = (value - minimum) / (maximum - minimum)
        fill.Size = UDim2.new(alpha, 0, 1, 0)
        valueLabel.Text = tostring(value)
        if not silent and callback then callback(value) end
    end
    handle.Get = function() return value end
    return handle
end

function Vexora:CreateDropdown(section, labelText, options, defaultValue, callback)
    local selected = defaultValue
    local expanded = false
    local optionConnections = {}
    local frame = create("Frame", {
        BackgroundColor3 = self.Theme.Surface2,
        Size = UDim2.new(1, 0, 0, 40),
        ClipsDescendants = true,
    }, section.Body)
    corner(frame, 8)
    local header = create("TextButton", {
        BackgroundTransparency = 1,
        Text = "",
        Size = UDim2.new(1, 0, 0, 40),
        AutoButtonColor = false,
    }, frame)
    local label = makeLabel(header, tostring(labelText), 12, self.Theme.Text)
    label.Position = UDim2.fromOffset(10, 0)
    label.Size = UDim2.new(0.48, -10, 1, 0)
    local selectedLabel = makeLabel(header, tostring(selected or "Select"), 11, self.Theme.Accent)
    selectedLabel.TextXAlignment = Enum.TextXAlignment.Right
    selectedLabel.Position = UDim2.new(0.48, 0, 0, 0)
    selectedLabel.Size = UDim2.new(0.52, -28, 1, 0)
    local arrow = makeLabel(header, "⌄", 16, self.Theme.Muted)
    arrow.TextXAlignment = Enum.TextXAlignment.Center
    arrow.Position = UDim2.new(1, -28, 0, 0)
    arrow.Size = UDim2.fromOffset(22, 40)
    local optionsHost = create("Frame", {
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(8, 42),
        Size = UDim2.new(1, -16, 0, 0),
    }, frame)
    create("UIListLayout", { Padding = UDim.new(0, 4) }, optionsHost)

    local function clearOptions()
        Vexora:DisconnectBucket(optionConnections)
        for _, child in ipairs(optionsHost:GetChildren()) do
            if child:IsA("TextButton") then child:Destroy() end
        end
    end
    local function refresh(newOptions)
        clearOptions()
        options = newOptions or options
        for _, option in ipairs(options) do
            local optionButton = create("TextButton", {
                Text = tostring(option),
                Font = Enum.Font.Gotham,
                TextSize = 11,
                TextColor3 = self.Theme.Text,
                BackgroundColor3 = self.Theme.Surface,
                Size = UDim2.new(1, 0, 0, 28),
                AutoButtonColor = false,
            }, optionsHost)
            corner(optionButton, 6)
            self:Connect(optionButton.MouseButton1Click, function()
                selected = option
                selectedLabel.Text = tostring(option)
                expanded = false
                frame:TweenSize(UDim2.new(1, 0, 0, 40), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.16, true)
                if callback then callback(selected) end
            end, optionConnections)
        end
    end
    self:Connect(header.MouseButton1Click, function()
        expanded = not expanded
        local height = expanded and (44 + #options * 32) or 40
        frame:TweenSize(UDim2.new(1, 0, 0, height), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.16, true)
    end)
    refresh(options or {})
    local handle = addWidgetHandle(frame, frame, optionConnections)
    handle.Refresh = refresh
    handle.Get = function() return selected end
    handle.Set = function(value, silent)
        selected = value
        selectedLabel.Text = tostring(value)
        if not silent and callback then callback(value) end
    end
    return handle
end

function Vexora:CreateButton(section, labelText, callback)
    local button = create("TextButton", {
        Text = tostring(labelText),
        Font = Enum.Font.GothamSemibold,
        TextSize = 12,
        TextColor3 = Color3.new(1, 1, 1),
        BackgroundColor3 = self.Theme.Accent,
        Size = UDim2.new(1, 0, 0, 38),
        AutoButtonColor = false,
    }, section.Body)
    corner(button, 8)
    local connections = {}
    self:Connect(button.MouseButton1Click, function()
        tween(button, 0.08, { BackgroundTransparency = 0.18 })
        task.delay(0.1, function()
            if button.Parent then tween(button, 0.12, { BackgroundTransparency = 0 }) end
        end)
        if callback then
            local ok, err = pcall(callback)
            if not ok then self:Notify("Button error: " .. tostring(err), "error") end
        end
    end, connections)
    return addWidgetHandle(button, button, connections)
end

function Vexora:CreateKeybind(section, labelText, defaultKey, callback)
    local key = defaultKey or Enum.KeyCode.Unknown
    local listening = false
    local row = create("Frame", {
        BackgroundColor3 = self.Theme.Surface2,
        Size = UDim2.new(1, 0, 0, 38),
    }, section.Body)
    corner(row, 8)
    local label = makeLabel(row, tostring(labelText), 12, self.Theme.Text)
    label.Position = UDim2.fromOffset(10, 0)
    label.Size = UDim2.new(1, -112, 1, 0)
    local bindButton = create("TextButton", {
        Text = key.Name,
        Font = Enum.Font.GothamMedium,
        TextSize = 11,
        TextColor3 = self.Theme.Accent,
        BackgroundColor3 = self.Theme.Surface,
        Size = UDim2.fromOffset(92, 26),
        Position = UDim2.new(1, -102, 0, 6),
        AutoButtonColor = false,
    }, row)
    corner(bindButton, 7)
    local connections = {}
    self:Connect(bindButton.MouseButton1Click, function()
        listening = true
        bindButton.Text = "Press key..."
    end, connections)
    self:Connect(UserInputService.InputBegan, function(input, processed)
        if processed then return end
        if listening and input.UserInputType == Enum.UserInputType.Keyboard then
            key = input.KeyCode
            bindButton.Text = key.Name
            listening = false
        elseif not listening and input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == key then
            if callback then callback(key) end
        end
    end, connections)
    local handle = addWidgetHandle(row, row, connections)
    handle.Get = function() return key end
    handle.Set = function(newKey)
        if typeof(newKey) == "EnumItem" and newKey.EnumType == Enum.KeyCode then
            key = newKey
            bindButton.Text = key.Name
        end
    end
    return handle
end

function Vexora:CreateColorPicker(section, labelText, defaultColor, callback)
    local color = typeof(defaultColor) == "Color3" and defaultColor or self.Theme.Accent
    local palette = {
        Color3.fromRGB(126, 105, 255),
        Color3.fromRGB(45, 185, 220),
        Color3.fromRGB(65, 190, 125),
        Color3.fromRGB(235, 105, 170),
        Color3.fromRGB(235, 170, 65),
        Color3.fromRGB(225, 75, 95),
    }
    local row = create("Frame", {
        BackgroundColor3 = self.Theme.Surface2,
        Size = UDim2.new(1, 0, 0, 40),
    }, section.Body)
    corner(row, 8)
    local label = makeLabel(row, tostring(labelText), 12, self.Theme.Text)
    label.Position = UDim2.fromOffset(10, 0)
    label.Size = UDim2.new(1, -170, 1, 0)
    local swatches = create("Frame", {
        BackgroundTransparency = 1,
        Position = UDim2.new(1, -158, 0, 8),
        Size = UDim2.fromOffset(148, 24),
    }, row)
    create("UIListLayout", {
        FillDirection = Enum.FillDirection.Horizontal,
        Padding = UDim.new(0, 5),
    }, swatches)
    local connections = {}
    local function setColor(nextColor, silent)
        if typeof(nextColor) ~= "Color3" then return end
        color = nextColor
        if not silent and callback then callback(color) end
    end
    for _, swatchColor in ipairs(palette) do
        local swatch = create("TextButton", {
            Text = "",
            BackgroundColor3 = swatchColor,
            Size = UDim2.fromOffset(20, 20),
            AutoButtonColor = false,
        }, swatches)
        corner(swatch, 10)
        self:Connect(swatch.MouseButton1Click, function() setColor(swatchColor) end, connections)
    end
    local handle = addWidgetHandle(row, row, connections)
    handle.Get = function() return color end
    handle.Set = setColor
    return handle
end

        if Main and Main.Parent then Main.Visible = false end
    end)
end

    if Main.Visible then closePanel() else openPanel() end

    if processed then return end

        if Main.Visible then closePanel() else openPanel() end
    end
end)

--// CLEAN TEARDOWN
function Vexora:Destroy()
    if self.Destroyed then return end
    self.Destroyed = true
    self:DisconnectBucket(self.Connections)
    for _, tab in pairs(self.Tabs) do
        if tab and tab.Destroy then pcall(function() tab:Destroy() end) end
    end
    for i = #self.CleanupTasks, 1, -1 do
        pcall(self.CleanupTasks[i])
        self.CleanupTasks[i] = nil
    end
    if ScreenGui and ScreenGui.Parent then
        ScreenGui:Destroy()
    end
end
